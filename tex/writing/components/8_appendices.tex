\addcontentsline{toc}{section}{Appendix 1: Solving the reflection problem}
\label{app-reflection-problem-instr-derivation}

\begin{align*}
    &\quad\ 
    \sum_{k=0}^{\infty} \theta_0^k \mathbf{W}^k (\mathbf{x}_t\boldsymbol{\beta}
    + \mathbf{W}\mathbf{x}_t \boldsymbol\beta \cdot \xi_0) 
    \\
    &= \sum_{k=0}^{\infty} \theta_0^k \mathbf{W}^k \mathbf{x}_t\boldsymbol{\beta} 
        + \sum_{k=0}^{\infty} \theta_0^k \mathbf{W}^{k+1} \mathbf{x}_t \boldsymbol\beta \cdot \xi_0
        \\
    &= \mathbf{x}_t\boldsymbol{\beta} 
        + \sum_{k=1}^{\infty} \theta_0^k \mathbf{W}^k \mathbf{x}_t\boldsymbol{\beta} 
        + \sum_{k=0}^{\infty} \theta_0^k \mathbf{W}^{k+1} \mathbf{x}_t \boldsymbol\beta \cdot \xi_0
        \\
    &= \mathbf{x}_t\boldsymbol{\beta} 
        + \sum_{k=0}^{\infty} \theta_0^k \mathbf{W}^{k+1} \mathbf{x}_t\boldsymbol{\beta} \cdot \theta_0
        + \sum_{k=0}^{\infty} \theta_0^k \mathbf{W}^{k+1} \mathbf{x}_t \boldsymbol\beta \cdot \xi_0
        \\
    &= \mathbf{x}_t\boldsymbol{\beta} 
        + \left( \sum_{k=0}^{\infty} \theta_0^k \mathbf{W}^{k+1} \mathbf{x}_t \right) 
        \boldsymbol\beta \cdot (\theta_0 + \xi_0)
        \\[1em]
    \mathbf{y}_t 
    &= \frac{\alpha}{1-\theta_0}\boldsymbol{\iota} 
        + \mathbf{x}_t\boldsymbol{\beta} 
        + \left( \sum_{k=0}^{\infty} \theta_0^k \mathbf{W}^{k+1} \mathbf{x}_t \right) 
        \boldsymbol\beta \cdot (\theta_0 + \xi_0) 
        + \sum_{k=0}^{\infty} \theta_0^k \mathbf{W}^k \boldsymbol{\varepsilon}_t
\end{align*}
