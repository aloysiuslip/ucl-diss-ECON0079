\newcommand{\wraptable}[3]{%
    % 1. Find the first segment (separated by _) containing a number,
    %    and remove everything up to and including that segment.
    \noexpandarg%
    \saveexpandmode%
    \expandarg%
    % Check for the first digit occurrence to identify the numbered segment
    \StrPosition{#2}{0}[\pos0]%
    \StrPosition{#2}{1}[\pos1]%
    \StrPosition{#2}{2}[\pos2]%
    \StrPosition{#2}{3}[\pos3]%
    \StrPosition{#2}{4}[\pos4]%
    \StrPosition{#2}{5}[\pos5]%
    \StrPosition{#2}{6}[\pos6]%
    \StrPosition{#2}{7}[\pos7]%
    \StrPosition{#2}{8}[\pos8]%
    \StrPosition{#2}{9}[\pos9]%
    % Extract everything after the underscore following the numbered segment
    \StrBehind{#2}{_}[\tempFirstPass]%
    \IfSubStr{\tempFirstPass}{_}{%
        \StrBehind{\tempFirstPass}{_}[\tempA]%
    }{%
        \let\tempA\tempFirstPass%
    }%
    \restoreexpandmode%
    %
    % 2. Replace remaining underscores with hyphens
    \StrSubstitute{\tempA}{_}{-}[\tempB]%
    \StrSubstitute{\tempB}{_}{-}[\tempC]%
    \edef\tablelabel{tab:\tempC}%
    %
    \begin{table}[hbtp]
        \centering
        \begin{threeparttable}
            \caption{#1}%
            \label{\tablelabel}% Automatically generated label
            \input{#2}%
            \ifblank{#3}{}{% Conditionally include notes[cite: 1, 4]
                \begin{tablenotes}
                #3
                \end{tablenotes}%
            }%
        \end{threeparttable}
    \end{table}%
}
