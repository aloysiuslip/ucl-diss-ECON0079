\newcommand{\wraptable}[3]{%
    \begin{table}[hbtp]
        \centering
        \begin{threeparttable}
            \caption{#1}%
            \input{#2}% Ensure this inputted file has NO \begin{center} or \end{center}
            \ifblank{#3}{}{% Only print tablenotes if #3 is non-empty
                \begin{tablenotes}
                #3
                \end{tablenotes}%
            }%
        \end{threeparttable}
    \end{table}%
}
